# 身份代理签名

第三方应用可以通过代理登录接口将用户映射到评论平台的内部账号，实现无缝登录体验。

## 工作原理

1. 第三方应用使用 `appId` 和 `secret` 生成 HMAC-SHA256 签名的 proxyToken
2. 用户携带 proxyToken 访问评论组件
3. 评论组件调用 `/api/platform/proxy-login` 验证签名并获取内部 JWT
4. nonce 防重放攻击，每个 token 只能使用一次

## Proxy Token 生成格式

```
proxyToken = base64url(JSON(payload)) + "." + hex(HMAC-SHA256(appSecret, base64url(JSON(payload))))
```

### Payload 结构

```json
{
  "externalId": "user_12345",
  "externalEmail": "user@example.com",
  "externalNick": "用户昵称",
  "nonce": "unique_random_string",
  "exp": 1700000000000
}
```

| 字段 | 类型 | 必填 | 说明 |
|---|---|---|---|
| externalId | string | 是 | 第三方平台的用户唯一标识 |
| externalEmail | string | 否 | 用户邮箱 |
| externalNick | string | 否 | 用户昵称 |
| nonce | string | 是 | 唯一随机字符串，防重放 |
| exp | number | 否 | 过期时间戳（毫秒） |

## Node.js 示例

```javascript
const crypto = require('crypto');

function generateProxyToken(appSecret, payload) {
  const encoded = Buffer.from(JSON.stringify(payload)).toString('base64url');
  const signature = crypto
    .createHmac('sha256', appSecret)
    .update(encoded)
    .digest('hex');
  return `${encoded}.${signature}`;
}

// 使用示例
const token = generateProxyToken('your-app-secret', {
  externalId: 'user_12345',
  externalEmail: 'user@example.com',
  externalNick: '张三',
  nonce: crypto.randomBytes(16).toString('hex'),
  exp: Date.now() + 5 * 60 * 1000, // 5分钟过期
});

console.log(token);
```

## Python 示例

```python
import hmac
import hashlib
import base64
import json
import time
import os

def generate_proxy_token(app_secret: str, payload: dict) -> str:
    encoded = base64.urlsafe_b64encode(
        json.dumps(payload, separators=(',', ':')).encode()
    ).decode().rstrip('=')

    signature = hmac.new(
        app_secret.encode(),
        encoded.encode(),
        hashlib.sha256
    ).hexdigest()

    return f"{encoded}.{signature}"

# 使用示例
token = generate_proxy_token('your-app-secret', {
    'externalId': 'user_12345',
    'externalEmail': 'user@example.com',
    'externalNick': '张三',
    'nonce': os.urandom(16).hex(),
    'exp': int(time.time() * 1000) + 5 * 60 * 1000,
})

print(token)
```

## PHP 示例

```php
<?php

function generateProxyToken(string $appSecret, array $payload): string {
    $encoded = rtrim(strtr(base64_encode(json_encode($payload, JSON_UNESCAPED_SLASHES)), '+/', '-_'), '=');

    $signature = hash_hmac('sha256', $encoded, $appSecret);

    return $encoded . '.' . $signature;
}

// 使用示例
$token = generateProxyToken('your-app-secret', [
    'externalId' => 'user_12345',
    'externalEmail' => 'user@example.com',
    'externalNick' => '张三',
    'nonce' => bin2hex(random_bytes(16)),
    'exp' => time() * 1000 + 5 * 60 * 1000,
]);

echo $token;
?>
```

## Java 示例

```java
import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.util.Base64;
import java.nio.charset.StandardCharsets;
import com.google.gson.Gson;

public class ProxyTokenGenerator {

    public static String generateProxyToken(String appSecret, Object payload) throws Exception {
        Gson gson = new Gson();
        String json = gson.toJson(payload);
        String encoded = Base64.getUrlEncoder().withoutPadding().encodeToString(
            json.getBytes(StandardCharsets.UTF_8)
        );

        Mac mac = Mac.getInstance("HmacSHA256");
        SecretKeySpec keySpec = new SecretKeySpec(
            appSecret.getBytes(StandardCharsets.UTF_8), "HmacSHA256"
        );
        mac.init(keySpec);
        byte[] hmacBytes = mac.doFinal(encoded.getBytes(StandardCharsets.UTF_8));

        StringBuilder hexString = new StringBuilder();
        for (byte b : hmacBytes) {
            hexString.append(String.format("%02x", b));
        }

        return encoded + "." + hexString;
    }

    public static void main(String[] args) throws Exception {
        Object payload = new Object() {
            public String externalId = "user_12345";
            public String externalEmail = "user@example.com";
            public String externalNick = "张三";
            public String nonce = java.util.UUID.randomUUID().toString().replace("-", "");
            public long exp = System.currentTimeMillis() + 5 * 60 * 1000;
        };

        String token = generateProxyToken("your-app-secret", payload);
        System.out.println(token);
    }
}
```

## 安全注意事项

1. **secret 绝不能泄露**：secret 用于生成签名，一旦泄露，攻击者可以伪造任何用户的登录请求
2. **nonce 必须唯一**：每次请求都生成新的 nonce，服务端会拒绝重复的 nonce
3. **设置合理的过期时间**：建议不超过 5 分钟
4. **使用 HTTPS**：所有通信必须使用 HTTPS 加密
